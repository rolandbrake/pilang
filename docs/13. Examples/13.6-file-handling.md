# 13.6 File Handling

The `fs` module works with files, paths, and directories.

```swift
import fs:f
```

File examples affect the real filesystem. Close file handles when you are done.

## Write and read a file

```swift
import fs:f

path = "notes.txt"

file = f.open(path, "w")
f.write(file, "first line\n")
f.write(file, "second line\n")
f.close(file)

file = f.open(path, "r")
text = f.read(file)
f.close(file)

println(text)
```

## Append to a file

```swift
import fs:f

file = f.open("notes.txt", "a")
f.append(file, "third line\n")
f.close(file)
```

## Check file information

```swift
import fs:f

path = "notes.txt"

if f.exists(path) {
    println("exists")
    println(f.abspath(path))
    println(f.basename(path))
    println(f.dirname(f.abspath(path)))
    println(f.ext(path))
    println(f.size(path))
}
```

## List a directory

```swift
import fs:f

for name in f.listdir(".") {
    println(name)
}
```

## Build paths safely

Use `fs.join` instead of manually writing path separators.

```swift
import fs:f

dir = "logs"
path = f.join(dir, "today.txt")

if !f.exists(dir) {
    f.mkdir(dir)
}

file = f.open(path, "w")
f.write(file, "log started\n")
f.close(file)
```

## Copy, move, and delete

```swift
import fs:f

f.copy("notes.txt", "notes-copy.txt")
f.move("notes-copy.txt", "archive.txt")

if f.exists("archive.txt") {
    f.delete("archive.txt")
}
```

## A small line counter

```swift
import fs:f
import string:s

fun count_lines(path) {
    let file = f.open(path, "r")
    let text = f.read(file)
    f.close(file)

    if text == "" {
        return 0
    }

    return len(s.split(text, "\n"))
}

println(count_lines("notes.txt"))
```

## What this example uses

- opening files
- reading and writing
- appending
- closing handles
- path helpers
- directory listing
- file copy, move, and delete
