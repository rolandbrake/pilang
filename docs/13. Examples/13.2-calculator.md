# 13.2 Calculator

This example builds a small calculator using functions and conditional dispatch.

```swift
fn add(a, b) {
    return a + b
}

fn sub(a, b) {
    return a - b
}

fn mult(a, b) {
    return a * b
}

fn div(a, b) {
    if b == 0 {
        error("division by zero")
    }

    return a / b
}

fn calculate(op, a, b) {
    if op == "+" {
        return add(a, b)
    } elif op == "-" {
        return sub(a, b)
    } elif op == "*" {
        return mult(a, b)
    } elif op == "/" {
        return div(a, b)
    }

    error("unknown operator: " + op)
}

println(calculate("+", 10, 5)) // 15
println(calculate("-", 10, 5)) // 5
println(calculate("*", 10, 5)) // 50
println(calculate("/", 10, 5)) // 2
```

## Calculator with a function map

Maps can store functions. This keeps the dispatch logic compact.

```swift
ops = {
    "+": fn (a, b) { return a + b },
    "-": fn (a, b) { return a - b },
    "*": fn (a, b) { return a * b },
    "/": fn (a, b) {
        if b == 0 {
            error("division by zero")
        }

        return a / b
    }
}

fn calculate(op, a, b) {
    if !(op in ops) {
        error("unknown operator: " + op)
    }

    return ops[op](a, b)
}

println(calculate("*", 6, 7)) // 42
```

## Adding power and square root

Use arithmetic operators and the `math` module.

```swift
import math:m

fn power(base, exponent) {
    return base ** exponent
}

fn distance(x, y) {
    return m.sqrt((x ** 2) + (y ** 2))
}

println(power(2, 8)) // 256
println(distance(3, 4)) // 5
```

## Named arguments

Named arguments make call sites easier to read.

```swift
fn percent(value, total) {
    if total == 0 {
        return 0
    }

    return (value / total) * 100
}

println(percent(value = 25, total = 200)) // 12.5
```
