# 13.7 Functional Programming

This example uses `map`, `filter`, `reduce`, and helpers from the `func` module.

## Filtering and mapping

```swift
numbers = [1, 2, 3, 4, 5, 6]

evens = filter(numbers, n -> n % 2 == 0)
squares = map(evens, n -> n * n)

println(squares) // [4, 16, 36]
```

## Reducing

```swift
numbers = [1, 2, 3, 4]

total = reduce(numbers, fun (acc, n) {
    return acc + n
}, 0)

println(total) // 10
```

## Data processing example

```swift
orders = [
    {"id": 1, "total": 25, "paid": true},
    {"id": 2, "total": 80, "paid": false},
    {"id": 3, "total": 40, "paid": true}
]

paid_orders = filter(orders, order -> order["paid"])
paid_totals = map(paid_orders, order -> order["total"])

revenue = reduce(paid_totals, fun (acc, total) {
    return acc + total
}, 0)

println(revenue) // 65
```

## Composition

```swift
import func:f

fun trim_name(name) {
    return trim(name)
}

fun normalize_name(name) {
    return upper(name)
}

clean_name = f.pipe(trim_name, normalize_name)

println(clean_name("  pilang  "))
```

`f.pipe` runs from left to right. `f.compose` runs from right to left.

```swift
import func:f

double_then_label = f.pipe(
    n -> n * 2,
    n -> "value: " + n
)

println(double_then_label(21)) // value: 42
```

## Partial application

```swift
import func:f

fun add(a, b) {
    return a + b
}

add10 = f.partial(add, 10)

println(add10(5)) // 15
```

## Currying

```swift
import func:f

fun volume(width, height, depth) {
    return width * height * depth
}

curried_volume = f.curry(volume)
area_10_by_20 = curried_volume(10)(20)

println(area_10_by_20(3)) // 600
```

## Memoization

```swift
import func:f

fun fib(n) {
    if n <= 1 {
        return n
    }

    return fib(n - 1) + fib(n - 2)
}

fast_fib = f.memoize(fib)

println(fast_fib(20))
println(fast_fib(20))
```

Memoization works best with pure functions.

## Lazy values with iterate

```swift
import func:f

next_power = f.iterate(1, n -> n * 2)

println(next_power()) // 1
println(next_power()) // 2
println(next_power()) // 4
println(next_power()) // 8
```
